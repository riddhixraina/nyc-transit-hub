erDiagram
    stations {
        TEXT stop_id PK
        TEXT stop_name
        REAL stop_lat
        REAL stop_lon
        TEXT parent_station
        INT ada_accessible
        TEXT borough
        TEXT daytime_routes
    }

    routes {
        TEXT route_id PK
        TEXT route_short_name
        TEXT route_long_name
        TEXT route_color
        TEXT route_text_color
    }

    trip_updates {
        INT id PK
        TEXT trip_id
        TEXT route_id FK
        TEXT stop_id FK
        TEXT arrival_time
        TEXT departure_time
        INT delay_seconds
        TEXT fetched_at
    }

    service_alerts {
        INT id PK
        TEXT alert_id UK
        TEXT header_text
        TEXT description_text
        TEXT severity_level
        TEXT effect
        TEXT affected_routes
        TEXT affected_stops
        TEXT active_period_start
        TEXT active_period_end
        TEXT fetched_at
    }

    elevator_equipment {
        TEXT equipment_id PK
        TEXT equipment_type
        TEXT station_name
        TEXT stop_id FK
        TEXT short_description
        INT ada_compliant
        TEXT lines_served
    }

    elevator_outages {
        INT id PK
        TEXT equipment_id FK
        TEXT reason
        TEXT out_of_service_date
        TEXT estimated_return
        INT is_upcoming
        TEXT fetched_at
    }

    stations ||--o{ trip_updates : "stop_id"
    routes ||--o{ trip_updates : "route_id"
    stations ||--o| elevator_equipment : "stop_id"
    elevator_equipment ||--o{ elevator_outages : "equipment_id"
